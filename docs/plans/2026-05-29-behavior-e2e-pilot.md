**Goal:** Stand up the Behavior-Driven E2E methodology
(`<workspace>/docs/process/behavior-e2e-capture.md`) in `pdomain-ocr-simple-gui`
as the Web/GUI pilot: authoritative per-screen behavior specs, behavior-
asserting Playwright tests (both observable-output **and** backend/side-
effect assertions), a two-tier execution model (deterministic + real
engine), and an ID-traceable generated coverage gate.

**Architecture:** Per-screen behavior records (`docs/specs/behavior/screen-*.md`)
are the source of truth. Each record has a stable ID (`B-<SCREEN>-NNN`).
Playwright tests cite those IDs and assert the record's observable output
(DOM/toasts) **and** its backend effect (re-query the API + inspect the
on-disk sidecar/`.txt`). Tier A runs against the existing fake-dispatcher
subprocess harness in CI; Tier B runs the real OCR engine on the GPU,
opt-in. A coverage script cross-checks declared vs cited IDs and gates CI.

**Tech Stack:** Python 3, FastAPI, pytest, pytest-playwright (Chromium),
pytest-xdist, uv, React/Vite/TS frontend, the existing `FakeStageDispatcher`
seam (`PDOMAIN_OCR_FAKE_DISPATCHER`), `PDOMAIN_GPU_BACKEND` for real-engine
selection.

# scripts/behavior_coverage.py
"""Behavior coverage audit: cross-check declared behavior IDs vs cited IDs.

Source of truth = docs/specs/behavior/*.md. Tests cite IDs via a
``Covers: B-...`` docstring line or a ``@behavior("B-...")`` marker.
"""
from __future__ import annotations

import re
import sys
from dataclasses import dataclass
from pathlib import Path

ID_RE = re.compile(r"\b([BF]-[A-Z0-9]+-\d+)\b")
RECORD_HEADING_RE = re.compile(r"^###\s+([BF]-[A-Z0-9]+-\d+)\b", re.MULTILINE)


@dataclass(frozen=True)
class Record:
    id: str
    regression: bool


def scan_declared(docs_dir: Path) -> dict[str, Record]:
    """Find every record/flow declared as an H3 heading, with its regression flag."""
    declared: dict[str, Record] = {}
    for md in sorted(docs_dir.glob("*.md")):
        if md.name in {"coverage.md", "README.md"}:
            continue
        text = md.read_text(encoding="utf-8")
        # Split into blocks starting at each H3 record heading.
        matches = list(RECORD_HEADING_RE.finditer(text))
        for i, m in enumerate(matches):
            rec_id = m.group(1)
            start = m.end()
            end = matches[i + 1].start() if i + 1 < len(matches) else len(text)
            block = text[start:end]
            regression = bool(
                re.search(r"\*\*Regression:\*\*\s*yes", block, re.IGNORECASE)
            )
            declared[rec_id] = Record(id=rec_id, regression=regression)
    return declared


def scan_cited(tests_dir: Path) -> set[str]:
    """Find every behavior ID cited in test files (docstring or marker)."""
    cited: set[str] = set()
    for py in tests_dir.rglob("*.py"):
        text = py.read_text(encoding="utf-8")
        for line in text.splitlines():
            if "Covers:" in line or "@behavior" in line:
                cited.update(ID_RE.findall(line))
    return cited
```

@dataclass(frozen=True)
class Report:
    declared: dict[str, Record]
    cited: set[str]
    orphans: set[str]
    unlinked: set[str]
    uncovered_regressions: set[str]

    @property
    def ok(self) -> bool:
        return not self.unlinked and not self.uncovered_regressions


def build_report(declared: dict[str, Record], cited: set[str]) -> Report:
    declared_ids = set(declared)
    orphans = declared_ids - cited
    unlinked = cited - declared_ids
    uncovered_regressions = {
        rid for rid in orphans if declared[rid].regression
    }
    return Report(
        declared=declared,
        cited=cited,
        orphans=orphans,
        unlinked=unlinked,
        uncovered_regressions=uncovered_regressions,
    )
```

```python
def render_markdown(report: Report) -> str:
    lines = [
        "# Behavior coverage (generated — do not edit)",
        "",
        "Run `make behavior-coverage` to regenerate.",
        "",
        "| ID | Regression | Status |",
        "|----|------------|--------|",
    ]
    for rid in sorted(report.declared):
        rec = report.declared[rid]
        status = "test-written" if rid in report.cited else "specified"
        reg = "yes" if rec.regression else "no"
        lines.append(f"| {rid} | {reg} | {status} |")
    if report.unlinked:
        lines += ["", "## Unlinked citations (FAIL — typo/stale)", ""]
        lines += [f"- {rid}" for rid in sorted(report.unlinked)]
    if report.uncovered_regressions:
        lines += ["", "## Uncovered regressions (FAIL)", ""]
        lines += [f"- {rid}" for rid in sorted(report.uncovered_regressions)]
    return "\n".join(lines) + "\n"


def main(argv: list[str] | None = None) -> int:
    root = Path(__file__).resolve().parent.parent
    docs_dir = root / "docs" / "specs" / "behavior"
    tests_dir = root / "tests"
    declared = scan_declared(docs_dir)
    cited = scan_cited(tests_dir)
    report = build_report(declared, cited)
    (docs_dir / "coverage.md").write_text(render_markdown(report), encoding="utf-8")
    if not report.ok:
        print("BEHAVIOR COVERAGE GATE FAILED", file=sys.stderr)
        if report.unlinked:
            print(f"  unlinked citations: {sorted(report.unlinked)}", file=sys.stderr)
        if report.uncovered_regressions:
            print(
                f"  uncovered regressions: {sorted(report.uncovered_regressions)}",
                file=sys.stderr,
            )
        return 1
    print(f"behavior coverage OK: {len(declared)} records, {len(cited)} cited")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

## Per-screen capture recipe (applies to M3–M6)

Each per-screen milestone repeats the **same recipe**. It is interview-
gated: do not invent intent — confirm it with the maintainer.

1. **Inventory (agent).** Read the screen's component, the backend
   endpoints it calls, its `data-testid`s (`frontend/src/lib/testids.ts`),
   and the existing `test_click_paths_<group>.py` for that screen. List
   every trigger and state transition.
2. **Draft (agent).** Pre-fill proposed behavior records (full schema)
   into `docs/specs/behavior/screen-<name>.md` for the obvious/mechanical
   behaviors, assigning IDs `B-<SCREEN>-NNN`. Leave `Test:` empty.
3. **Interview (with maintainer).** Walk the draft; capture ambiguous
   behaviors, edge cases, failure paths, and known regressions
   (`Regression: yes (#ref)`). Lock the records.
4. **Tests (agent).** For each locked record, write/UPGRADE a Tier A
   Playwright test that (a) cites the ID in its docstring (`Covers: B-...`),
   (b) asserts the **observable output**, and (c) asserts the **backend
   effect** by re-querying the API **and** inspecting the on-disk artifact
   under `e2e_data_root`. Cover the good path and at least one bad path.
   Prefer upgrading the matching `test_click_paths_<group>.py` test in
   place over adding a parallel file.
5. **Regenerate + verify.**

   ```bash
   uv run python -m scripts.behavior_coverage
   make e2e-fast
   ```

6. **Commit** the screen doc + tests together.

### Reference test (the exact Tier A pattern to replicate)

This is the concrete shape every Tier A behavior test takes — observable

```python
# tests/e2e/test_click_paths_page_viewer.py (upgraded example)
import json
from pathlib import Path

import pytest
from playwright.sync_api import Page, expect

pytestmark = [pytest.mark.slow, pytest.mark.e2e]


def test_save_edited_text_persists(
    page: Page, live_server_url: str, e2e_data_root: Path, seeded_job_id: str
) -> None:
    """Covers: B-PAGEVIEW-012 — saving edited text updates sidecar + combined txt."""
    page.goto(f"{live_server_url}/jobs/{seeded_job_id}/pages/0")
    editor = page.get_by_test_id("page-text-editor")  # exact id from testids.ts
    expect(editor).to_be_visible(timeout=15_000)

    editor.fill("corrected text")
    page.keyboard.press("Control+s")

    # Observable output: a "Saved" toast appears.
    page.wait_for_function(
        "document.querySelectorAll('[data-sonner-toast]').length > 0"
    )

    # Backend effect #1 — API re-query returns the new text.
    import httpx
    resp = httpx.get(f"{live_server_url}/api/pages/{seeded_job_id}/0")
    assert resp.status_code == 200
    assert resp.json()["text"] == "corrected text"

    # Backend effect #2 — the on-disk sidecar reflects the edit.
    sidecar = e2e_data_root / "projects" / seeded_job_id / "page_0000.json"
    assert json.loads(sidecar.read_text())["text"] == "corrected text"


def test_save_edited_text_bad_state(
    page: Page, live_server_url: str, e2e_data_root: Path, seeded_job_id: str
) -> None:
    """Covers: B-PAGEVIEW-012 (bad path) — invalid page index does not mutate disk."""
    import httpx
    resp = httpx.put(
        f"{live_server_url}/api/pages/{seeded_job_id}/9999/text",
        json={"text": "x"},
    )
    assert resp.status_code == 404
    # No stray sidecar created for the bad index.
    assert not (e2e_data_root / "projects" / seeded_job_id / "page_9999.json").exists()
```

> assert observable output, re-query API, inspect disk, plus a bad path —
